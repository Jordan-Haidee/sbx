# TD7 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a production-quality `TD7` algorithm to `sbx`, including SALE, fixed embeddings, value clipping, LAP replay, and checkpoint-policy training/evaluation with delayed training pulses.

**Architecture:** Implement `TD7` as a new standalone off-policy algorithm under `sbx/td7`, reusing `sbx` JAX policy/train-state patterns where practical while owning its custom replay buffer and custom episode-driven learning lifecycle. Keep the public API aligned with `TD3`, but move TD7-specific rollout, checkpoint state, and delayed update logic into the new algorithm class to avoid polluting `TD3`.

**Tech Stack:** Python 3.10+, JAX, Flax, Optax, Gymnasium, Stable-Baselines3 replay/env utilities, pytest.

`sbx/td7/replay_buffer.py`

```python
from dataclasses import dataclass

import numpy as np


@dataclass
class TD7ReplayBufferSamples:
    observations: np.ndarray
    actions: np.ndarray
    next_observations: np.ndarray
    rewards: np.ndarray
    dones: np.ndarray
    indices: np.ndarray


class TD7ReplayBuffer:
    def __init__(self, buffer_size: int, observation_dim: int, action_dim: int, batch_size: int, alpha: float = 0.6):
        self.buffer_size = buffer_size
        self.batch_size = batch_size
        self.alpha = alpha
        self.pos = 0
        self.size = 0
        self.max_priority = 1.0
        self.observations = np.zeros((buffer_size, observation_dim), dtype=np.float32)
        self.actions = np.zeros((buffer_size, action_dim), dtype=np.float32)
        self.next_observations = np.zeros((buffer_size, observation_dim), dtype=np.float32)
        self.rewards = np.zeros((buffer_size,), dtype=np.float32)
        self.dones = np.zeros((buffer_size,), dtype=np.float32)
        self.priorities = np.ones((buffer_size,), dtype=np.float32)

    def add(self, observation, action, next_observation, reward, done):
        self.observations[self.pos] = observation
        self.actions[self.pos] = action
        self.next_observations[self.pos] = next_observation
        self.rewards[self.pos] = reward
        self.dones[self.pos] = float(done)
        self.priorities[self.pos] = self.max_priority
        self.pos = (self.pos + 1) % self.buffer_size
        self.size = min(self.size + 1, self.buffer_size)

    def sample(self, batch_size: int | None = None):
        batch_size = batch_size or self.batch_size
        scaled = self.priorities[: self.size] ** self.alpha
        probs = scaled / scaled.sum()
        indices = np.random.choice(self.size, size=batch_size, replace=True, p=probs)
        return TD7ReplayBufferSamples(
            observations=self.observations[indices],
            actions=self.actions[indices],
            next_observations=self.next_observations[indices],
            rewards=self.rewards[indices],
            dones=self.dones[indices],
            indices=indices,
        )

    def update_priorities(self, indices: np.ndarray, priorities: np.ndarray) -> None:
        self.priorities[indices] = priorities
        self.max_priority = max(self.max_priority, float(np.max(priorities)))

    def reset_max_priority(self) -> None:
        self.max_priority = float(np.max(self.priorities[: self.size]))
```

The final implementation may adapt constructor signatures to `sbx` needs, but must preserve these behaviors.

import jax.numpy as jnp

from gymnasium import spaces

def avg_l1_norm(x: jnp.ndarray, eps: float = 1e-8) -> jnp.ndarray:
    return x / jnp.clip(jnp.mean(jnp.abs(x), axis=-1, keepdims=True), a_min=eps)


class StateEncoder(nn.Module):
    hidden_dim: int = 256
    zs_dim: int = 256

    @nn.compact
    def __call__(self, obs: jnp.ndarray) -> jnp.ndarray:
        x = Flatten()(obs)
        x = nn.elu(nn.Dense(self.hidden_dim)(x))
        x = nn.elu(nn.Dense(self.hidden_dim)(x))
        return avg_l1_norm(nn.Dense(self.zs_dim)(x))

import numpy as np

from sbx import TD7


class TinyEpisodeEnv(gym.Env):
    def __init__(self):
        self.observation_space = gym.spaces.Box(-1.0, 1.0, shape=(3,))
        self.action_space = gym.spaces.Box(-1.0, 1.0, shape=(1,))
